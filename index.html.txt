    const THRESHOLDS = {
      atencao: 3.21,
      alerta: 3.71,
      emergencia: 4.2,
      extravasamento: 4.78
    };

    const riverPanel = document.getElementById("riverPanel");
    const riverLevel = document.getElementById("riverLevel");
    const riverStatus = document.getElementById("riverStatus");
    const lastUpdate = document.getElementById("lastUpdate");

    function classificarNivel(nivel) {
      if (nivel >= THRESHOLDS.extravasamento) return { classe: "extravasamento", texto: "Extravasamento" };
      if (nivel >= THRESHOLDS.emergencia) return { classe: "emergencia", texto: "Emergência" };
      if (nivel >= THRESHOLDS.alerta) return { classe: "alerta", texto: "Alerta" };
      if (nivel >= THRESHOLDS.atencao) return { classe: "alerta", texto: "Atenção" };
      return { classe: "", texto: "Normal" };
    }

    function atualizarInterface(nivel) {
      const status = classificarNivel(nivel);
      riverPanel.className = "river-panel" + (status.classe ? " " + status.classe : "");
      riverLevel.textContent = nivel.toFixed(2);
      riverStatus.textContent = status.texto;
      lastUpdate.textContent = new Date().toLocaleString("pt-BR");
    }

    async function buscarNivelAtual() {
      try {
        const nivelSimulado = 2.45;
        atualizarInterface(nivelSimulado);
      } catch (error) {
        console.error("Falha ao atualizar dados:", error);
        riverStatus.textContent = "Erro ao carregar dados";
      }
    }

    function renderGraficoNivel() {
      const x = ["Dia 1", "Dia 2", "Dia 3", "Dia 4", "Dia 5", "Dia 6", "Dia 7"];
      const y = [2.45, 2.52, 2.61, 2.74, 2.88, 2.95, 3.02];

        { type: "line", xref: "paper", x0: 0, x1: 1, y0: 4.20, y1: 4.20, line: { color: "#f97316", width: 2, dash: "dash" } },
        { type: "line", xref: "paper", x0: 0, x1: 1, y0: 4.78, y1: 4.78, line: { color: "#dc2626", width: 2, dash: "dash" } }
      ];

        { x: 1, y: 4.78, xref: "paper", yref: "y", text: "Extravasamento 4,78 m", showarrow: false, font: { color: "#dc2626" }, xanchor: "left", yanchor: "bottom" }

    async function carregarPrevisaoTempo() {
      const lat = -22.7269;
      const lon = -47.6492;
      const url = `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&daily=weather_code,temperature_2m_max,temperature_2m_min,precipitation_sum&timezone=America/Sao_Paulo&forecast_days=7`;

      const resp = await fetch(url);
      const data = await resp.json();

      const container = document.getElementById("previsaoTempo");
      const dias = data.daily.time;

      const icones = {
        0: "☀️",
        1: "🌤️",
        2: "⛅",
        3: "☁️",
        45: "🌫️",
        48: "🌫️",
        51: "🌦️",
        53: "🌦️",
        55: "🌧️",
        61: "🌧️",
        63: "🌧️",
        65: "⛈️",
        80: "🌦️",
        81: "🌧️",
        82: "⛈️",
        95: "⛈️"
      };

      container.innerHTML = dias.map((dia, i) => {
        const code = data.daily.weather_code[i];
        const icon = icones[code] || "🌡️";
        return `
          <div class="forecast-day">
            <h4>${new Date(dia + "T00:00:00").toLocaleDateString("pt-BR", {
              weekday: "short",
              day: "2-digit",
              month: "2-digit"
            })}</h4>
            <p style="font-size: 2rem;">${icon}</p>
            <p><strong>Mín:</strong> ${data.daily.temperature_2m_min[i]}°C</p>
            <p><strong>Máx:</strong> ${data.daily.temperature_2m_max[i]}°C</p>
            <p><strong>Chuva:</strong> ${data.daily.precipitation_sum[i]} mm</p>
          </div>
        `;
      }).join("");
    }

    buscarNivelAtual();
    renderGraficoNivel();
    carregarPrevisaoTempo();
    setInterval(buscarNivelAtual, 15 * 60 * 1000);
